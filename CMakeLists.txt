cmake_minimum_required(VERSION 3.24)
cmake_policy(SET CMP0091 NEW)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
project(GhostWaveStudio VERSION 2.0.0 LANGUAGES C CXX)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Build on the Windows x64 GitHub Actions runner with MSVC")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
set(wxBUILD_SHARED OFF CACHE BOOL "" FORCE)
set(wxBUILD_SAMPLES OFF CACHE BOOL "" FORCE)
set(wxBUILD_TESTS OFF CACHE BOOL "" FORCE)
set(wxBUILD_PRECOMP OFF CACHE BOOL "" FORCE)
set(wxUSE_ACCESSIBILITY ON CACHE BOOL "" FORCE)
add_subdirectory(vendor/wxWidgets EXCLUDE_FROM_ALL)
set(UTF8PROC_INSTALL OFF CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
add_subdirectory(vendor/utf8proc EXCLUDE_FROM_ALL)
add_library(ghostwave_core src/platform.cpp src/profile.cpp src/lyrics.cpp src/audio.cpp src/services.cpp src/cli.cpp)
target_include_directories(ghostwave_core PUBLIC include vendor/json/single_include PRIVATE vendor/pocketfft vendor/libsndfile/include)
target_compile_definitions(ghostwave_core PUBLIC NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00 UTF8PROC_STATIC)
target_compile_options(ghostwave_core PRIVATE /W4 /utf-8 /fp:strict /permissive- /EHsc)
target_link_libraries(ghostwave_core PUBLIC wx::base utf8proc PRIVATE bcrypt winhttp crypt32)
add_executable(GhostWaveStudio WIN32 src/gui.cpp resources/app.rc)
target_link_libraries(GhostWaveStudio PRIVATE ghostwave_core wx::core shell32)
target_compile_options(GhostWaveStudio PRIVATE /W4 /utf-8 /permissive- /EHsc)
add_executable(ghostwave-cli src/console.cpp)
target_link_libraries(ghostwave-cli PRIVATE ghostwave_core shell32)
target_compile_options(ghostwave-cli PRIVATE /utf-8 /EHsc)
enable_testing()
add_executable(ghostwave-tests tests/core_tests.cpp)
target_link_libraries(ghostwave-tests PRIVATE ghostwave_core)
target_compile_options(ghostwave-tests PRIVATE /W4 /utf-8 /fp:strict /EHsc)
add_test(NAME core COMMAND ghostwave-tests "${CMAKE_SOURCE_DIR}/ci-output/reference")
set_tests_properties(core PROPERTIES WORKING_DIRECTORY "${CMAKE_BINARY_DIR}/bin/Release")
foreach(target GhostWaveStudio ghostwave-cli ghostwave-tests)
  add_custom_command(TARGET ${target} POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_SOURCE_DIR}/resources" "$<TARGET_FILE_DIR:${target}>/resources"
    COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_SOURCE_DIR}/locales" "$<TARGET_FILE_DIR:${target}>/locales"
    COMMAND ${CMAKE_COMMAND} -E copy_if_different "${CMAKE_SOURCE_DIR}/vendor/media/ffmpeg.exe" "$<TARGET_FILE_DIR:${target}>/ffmpeg.exe"
    COMMAND ${CMAKE_COMMAND} -E copy_if_different "${CMAKE_SOURCE_DIR}/vendor/media/libsndfile_x64.dll" "$<TARGET_FILE_DIR:${target}>/libsndfile_x64.dll")
endforeach()
